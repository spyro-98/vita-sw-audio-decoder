cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 if(NOT DEFINED CMAKE_TOOLCHAIN_FILE AND DEFINED ENV{VITASDK})
  set(CMAKE_TOOLCHAIN_FILE "$ENV{VITASDK}/share/vita.toolchain.cmake" CACHE PATH "VitaSDK toolchain")
 endif()
 project(vita_sw_audio_decoder VERSION 1.0.0 LANGUAGES C)
endif()
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
if(NOT TARGET VitaAudioDecoder::Interface)
 set(VITA_AUDIO_DECODER_INTERFACE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../vita-hw-audio-decoder" CACHE PATH "Shared audio contracts")
 add_subdirectory("${VITA_AUDIO_DECODER_INTERFACE_ROOT}/interface" "${CMAKE_CURRENT_BINARY_DIR}/audio-interface")
endif()
set(VITA_SW_AUDIO_DECODER_FFMPEG_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../VitaMediaDeck/build/deps/ffmpeg-vita-hw" CACHE PATH "Pinned FFmpeg installation")
if(NOT EXISTS "${VITA_SW_AUDIO_DECODER_FFMPEG_ROOT}/lib/libswresample.a")
 message(FATAL_ERROR "Set VITA_SW_AUDIO_DECODER_FFMPEG_ROOT to the FFmpeg 6 prefix with AC3, MP3 and swresample")
endif()
add_library(vita_sw_audio_decoder STATIC src/vita_sw_audio_decoder.c src/module_log.c)
add_library(VitaSwAudioDecoder::VitaSwAudioDecoder ALIAS vita_sw_audio_decoder)
set_target_properties(vita_sw_audio_decoder PROPERTIES EXPORT_NAME VitaSwAudioDecoder VERSION 1.0.0)
target_include_directories(vita_sw_audio_decoder PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_compile_features(vita_sw_audio_decoder PUBLIC c_std_11)
target_compile_options(vita_sw_audio_decoder PRIVATE -mcpu=cortex-a9 -mfpu=neon -mfloat-abi=hard -O3 -ffunction-sections -fdata-sections)
target_include_directories(vita_sw_audio_decoder SYSTEM PUBLIC $<BUILD_INTERFACE:${VITA_SW_AUDIO_DECODER_FFMPEG_ROOT}/include>)
target_link_directories(vita_sw_audio_decoder PUBLIC $<BUILD_INTERFACE:${VITA_SW_AUDIO_DECODER_FFMPEG_ROOT}/lib>)
target_link_libraries(vita_sw_audio_decoder PUBLIC VitaAudioDecoder::Interface avcodec swresample avutil m)
install(TARGETS vita_sw_audio_decoder EXPORT VitaSwAudioDecoderTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT VitaSwAudioDecoderTargets NAMESPACE VitaSwAudioDecoder:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/VitaSwAudioDecoder)
configure_package_config_file(cmake/VitaSwAudioDecoderConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/VitaSwAudioDecoderConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/VitaSwAudioDecoder)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/VitaSwAudioDecoderConfigVersion.cmake VERSION 1.0.0 COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/VitaSwAudioDecoderConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/VitaSwAudioDecoderConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/VitaSwAudioDecoder)
install(FILES LICENSE THIRD_PARTY_NOTICES.md DEPENDENCIES.lock DESTINATION share/licenses/vita-sw-audio-decoder)
